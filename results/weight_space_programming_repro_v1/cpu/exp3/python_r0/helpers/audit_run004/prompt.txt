def cent(a):

g = W["ln1_w.0"].astype(float)

for j in range(6):

    v = cent(W["W_v.0"][sl].astype(float)*g)
    B.append(W["W_o.0"][:,sl].astype(float)@v)

_, _, vt = np.linalg.svd(np.vstack([B[1],B[5]]),full_matrices=False)
C = vt[:27].T
f = cent(W["W_fc.0"].astype(float)*W["ln2_w.0"].astype(float))
X = [f@C, f@B[5]@C, f@B[1]@C]

print("Principal-angle cosines between L0H1/H5 top-27 value-input spaces:")
r1 = np.linalg.svd(B[1],full_matrices=False)[2][:27].T
r5 = np.linalg.svd(B[5],full_matrices=False)[2][:27].T
print(np.linalg.svd(r1.T@r5,compute_uv=False))

# Fit only to actual weight rows and their linear-algebra compositions.
# No synthetic residual inputs or nonlinear network operations are used.
joint = np.vstack([x/np.linalg.norm(x)*np.sqrt(x.size) for x in X])
mu = joint.mean(axis=0)
ev, u = np.linalg.eigh(np.cov(joint,rowvar=False))
whitener = (u*(1/np.sqrt(np.maximum(ev,ev.max()*1e-3))))@u.T
Z = (joint-mu)@whitener
Zn = Z/np.maximum(np.linalg.norm(Z,axis=1,keepdims=True),1e-12)
km = KMeans(n_clusters=28,n_init=15,random_state=42).fit(Zn)
labels = km.labels_.reshape(3,1536)
centers = km.cluster_centers_
centers /= np.linalg.norm(centers,axis=1,keepdims=True)
assigned_cos = np.sum(Zn*centers[km.labels_],axis=1).reshape(3,1536)

print("\nWeight-row clustering quality, current / via H5 / via H1:")
for i in range(3):
